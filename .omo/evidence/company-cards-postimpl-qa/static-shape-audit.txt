import type { ResearchIndex } from "../../data/research"
import { BUSINESS_MODEL_LABELS } from "../../domain/display-labels"
import type { Company } from "../../domain/entity-schema"
import { CorpusClaim } from "../shared/CorpusClaim"

type CompanyDossierProps = {
  readonly company: Company
  readonly index: ResearchIndex
}

export function CompanyDossier({ company, index }: CompanyDossierProps) {
  return (
    <article className="landscape-record" id={`company-${company.companyId}`}>
      <p className="kicker">Named dossier · {company.entityStatus.replaceAll("_", " ")}</p>
      <h4>{company.name}</h4>
      <p className="evidence-text">
        {company.primarySegment.replaceAll("_", " ")} ·{" "}
        {company.businessModelIds.map((model) => BUSINESS_MODEL_LABELS[model]).join(" · ")}
      </p>
      <a href={company.canonicalDomain} rel="noreferrer" target="_blank">
        {new URL(company.canonicalDomain).hostname}
      </a>
      <CorpusClaim claimId={company.identityClaimId} index={index} />
      {company.claimGroups.currentOffer.map((claimId) => (
        <CorpusClaim claimId={claimId} index={index} key={claimId} />
      ))}
      {company.claimGroups.milestones.map((claimId) => (
        <CorpusClaim claimId={claimId} index={index} key={claimId} />
      ))}
      {company.claimGroups.risks.map((claimId) => (
        <CorpusClaim claimId={claimId} index={index} key={claimId} />
      ))}
    </article>
  )
}

import { researchIndex } from "../../data/research"
import type { ResearchFilters } from "../../domain/selectors"
import { selectAdjacent, selectCompanies } from "../../domain/selectors"
import { CorpusClaim } from "../shared/CorpusClaim"
import { SectionIntro } from "../shared/SectionIntro"
import { CompanyDossier } from "./CompanyDossier"
import { CoreLandscape } from "./CoreLandscape"
import { SupplementalRegister } from "./SupplementalRegister"

export function CompaniesPage({ filters = {} }: { readonly filters?: ResearchFilters }) {
  const companies = selectCompanies(researchIndex, filters)
  const adjacent = selectAdjacent(researchIndex, filters)
  return (
    <section className="feature-page">
      <SectionIntro eyebrow="04 · Company intelligence" title="Every company, one card.">
        <p>
          Ten named companies and the wider adjacent landscape, one card each. Post-cutoff
          discoveries are listed separately at the end.
        </p>
      </SectionIntro>
      <section aria-labelledby="named-company-title">
        <header className="section-heading-row">
          <div>
            <p className="kicker">Named-company dossiers</p>
            <h3 id="named-company-title">{companies.length} of 10 matches</h3>
          </div>
          <p>
            One card per company: identity, current offer, milestones, and risks, each with its
            sources.
          </p>
        </header>
        <div className="landscape-grid">
          {companies.map((company) => (
            <CompanyDossier company={company} index={researchIndex} key={company.companyId} />
          ))}
        </div>
      </section>
      <CoreLandscape index={researchIndex} records={adjacent} />
      <SupplementalRegister query={filters.q} />
    </section>
  )
}

export function renderClaimScope() {
  return (
    <div data-claim-scope="companies">
      {researchIndex.corpus.companies.map((company) => (
        <CorpusClaim
          claimId={company.identityClaimId}
          index={researchIndex}
          key={company.companyId}
        />
      ))}
    </div>
  )
}
